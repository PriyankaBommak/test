	@Override
	public void checkToCommit(String pActionType, ContrlAirspaceCommTriggerRequestDto reqDto) throws Exception {
		log.info("checkToCommit Executing");

		
		try {
			Integer vButton = 0;
			Integer totalRows = 0;
			String vButtonText = null;
			Integer msgnum = 0;

			if (Objects.equals(system.getFormStatus(), FormConstant. CHANGED)) {
				if (Objects.equals(pActionType, "COMMIT")) {
					vButtonText = "Cancel";

				}

				else if (Objects.equals(pActionType, "EXIT")) {
					vButtonText = "Exit Without Save";

				}

				else {
					vButtonText = "Cancel Modification";

				}
				if (Objects.equals(global.getLibRefreshed(), Y)) {
					alertDetails.getCurrent();
					if (alertDetails.getAlerts().size() < alertDetails.getCurrentAlert()) {
						vButton = displayAlert.moreButtons(S, "Refresh Record",
								"You have modified record(s). Select an option:" + "\n" + "\n"
										+ "1. Save and refresh Master Library" + "\n"
										+ "2. Cancel modification, NO Save, NO Refresh" + "\n",
								"Save&Refresh", vButtonText, "");
						alertDetails.createNewRecord("checkToCommit1");
						throw new AlertException(event, alertDetails);
					} else {
						vButton = alertDetails.getAlertValue("checkToCommit1", alertDetails.getCurrentAlert());
					}
					if (Integer.valueOf(1).equals(vButton)) {

						Integer ind = system.getCursorRecordIndex();
						validateBeforeCommit(reqDto);
						system.setCursorRecordIndex(ind);
						commitForm(this);
						sendUpdatedRowIdDetails();
						system.setFormStatus(FormConstant.QUERIED);
						parameter.setUpdRec(N);
						setBlockProperty(nameIn(this, "system.Cursor_Block", String.class), FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_TRUE);
						message("Record has been saved successfully");
						system.setFormStatus("NORMAL");
						totalRows = getGroupRowCount(findGroup("refreshRecordsGroup"));
						if (totalRows > 0) {
							refreshMasterLib.refreshRecords(totalRows);
						}

						if ((Objects.equals(msgnum, 40400))) {
							clearMessage();
						}

					}
				}

				else {
					if (Objects.equals(pActionType, "COMMIT")) {
						vButton = 1;

					}

					else {
						alertDetails.getCurrent();
						if (alertDetails.getAlerts().size() < alertDetails.getCurrentAlert()) {
							vButton = displayAlert.moreButtons(S, "Refresh Record",
									"Record is modified or inserted or deleted. Select an option:  " + "\n", "Save",
									vButtonText, "");
							alertDetails.createNewRecord("alertid");
							throw new AlertException(event, alertDetails);
						} else {
							vButton = alertDetails.getAlertValue("alertid", alertDetails.getCurrentAlert());
						}
					}
					if (Objects.equals(vButton, 1)) {
						Integer ind = system.getCursorRecordIndex();
						validateBeforeCommit(reqDto);
						system.setCursorRecordIndex(ind);
						commitForm(this);
						system.setFormStatus(FormConstant.QUERIED);
						parameter.setUpdRec(N);
						setBlockProperty(nameIn(this, "system.Cursor_Block", String.class), FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_TRUE);
						message("Record has been saved successfully");
						system.setFormStatus("NORMAL");
						if ((Objects.equals(msgnum, 40400))) {
							clearMessage();
						}
					} else {

						exitForm();

					}

				}
				if ((Objects.equals(global.getLibRefreshed(), Y) && Objects.equals(vButton, 2))
						|| (!Objects.equals(global.getLibRefreshed(), Y) && Objects.equals(vButton, 1))) {
					if (Objects.equals(pActionType, "COMMIT")) {
						log.debug("pActionType is Commit");
					} else if (Objects.equals(pActionType, "EXIT")) {

						setApplicationProperty("cursor_style", "cursor_style");

						exitForm();

					}

					else {

						clearBlock("noCommit", " ");

					}

				}

				else {
					if (Objects.equals(system.getFormStatus(), FormConstant. CHANGED)) {
						throw new FormTriggerFailureException();

					} else if (Objects.equals(pActionType, "EXIT")) {

						setApplicationProperty("cursorStyle", "default");
						exitForm();
					}

				}

			}

			else {

				refreshMasterLib.ifRefresh();
				if (pActionType.equals("EXIT")) {
					exitForm();
				}

			}

			log.info("checkToCommit Executed Successfully");
		} catch (Exception e) {
			if (e instanceof DataIntegrityViolationException || e instanceof NonUniqueObjectException
					|| e instanceof EntityExistsException || e instanceof ConstraintViolationException
					|| e instanceof org.hibernate.exception.ConstraintViolationException) {
				coreptLib.dspMsg("ORACLE error: unable to INSERT record. \n \n Please check the exact "
						+ "error message from the \"Display \n Error\" in the \"HELP\" menu");
				log.info(" Unique Constrain Error while Executing the keyCommit Service");
				throw new FormTriggerFailureException(e.getMessage());
			}

			log.error("Error while executing checkToCommit{}", e.getMessage());
			throw e;

		}
	}


	@Override
	public void doValidate(String pBlock, String pIgnoreRef) throws Exception {
		log.info("doValidate Executing");
	
		try {
			PlStdContrlAirspaceCommType vSrec ;
			
			String vAllErr = null;
			String vNerr = null;
			String vClass = null;
			String vInd = Y;

			if (!Objects.equals(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirspaceType(),
					null)) {
				controlBlock.setStdOverrideErrors(null);
				vClass = plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCommunicationClass();
				if (!Arrays.asList("E", "A").contains(vClass)) {
					if (!coreptLib.isOverride(
							global.getDataSupplier(), toInteger(plStdContrlAirspaceCommMr
									.getRow(system.getCursorRecordIndex()).getProcessingCycle()),
							"CONTRL_AIRSPACE_COMM", 979)) {
						vAllErr = "* 979 - " + coreptLib.getErrText(979);

						vInd = I;

					}

					else {
						if (!Objects.equals(vInd, I)) {
							vInd = O;

						}

					}
					if ((Objects.equals(
							rtrim(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent()),
							null)
							&& Objects.equals(rtrim(
									plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirIdent()),
									null))
							|| (!Objects.equals(rtrim(
									plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent()),
									null)
									&& !Objects.equals(rtrim(plStdContrlAirspaceCommMr
											.getRow(system.getCursorRecordIndex()).getFirUirIdent()), null))) {
						if (!coreptLib.isOverride(
								global.getDataSupplier(), toInteger(plStdContrlAirspaceCommMr
										.getRow(system.getCursorRecordIndex()).getProcessingCycle()),
								"CONTRL_AIRSPACE_COMM", 980)) {
							vAllErr = getNullClean(vAllErr) + " * 980 - " + coreptLib.getErrText(980);
							vInd = I;

						}

						else {
							if (!Objects.equals(vInd, I)) {
								vInd = O;

							}

						}

					}

				}

				else if (Objects.equals(vClass, "A") && (Objects.equals(
						rtrim(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent()), null)
						|| !Objects.equals(
								rtrim(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirIdent()),
								null))) {
					if (!coreptLib.isOverride(
							global.getDataSupplier(), toInteger(plStdContrlAirspaceCommMr
									.getRow(system.getCursorRecordIndex()).getProcessingCycle()),
							"CONTRL_AIRSPACE_COMM", 981)) {
						vAllErr = "* 981 - " + coreptLib.getErrText(981);
						vInd = I;

					}

					else {
						if (!Objects.equals(vInd, I)) {
							vInd = O;

						}

					}

				}

				else if (Objects.equals(vClass, "E") && (!Objects.equals(
						rtrim(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent()), null)
						|| Objects.equals(
								rtrim(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirIdent()),
								null))) {
					if (!coreptLib.isOverride(
							global.getDataSupplier(), toInteger(plStdContrlAirspaceCommMr
									.getRow(system.getCursorRecordIndex()).getProcessingCycle()),
							"CONTRL_AIRSPACE_COMM", 982)) {
						vAllErr = "* 982 - " + coreptLib.getErrText(982);
						vInd = I;

					}

					else {
						if (!Objects.equals(vInd, I)) {
							vInd = O;

						}

					}

				}

				Map<String, Object> populateRelRecordCall = populateRelRecord(pBlock);
				vSrec = (PlStdContrlAirspaceCommType) populateRelRecordCall.get("pSrec");
				Map<String, Object> dbCalls = app.executeProcedure("CPTS", "vcontrl_airspace_comm",
						"RECRV3_WRAPPER",
						new ProcedureInParameter("p_record_type", S, OracleTypes.VARCHAR),
						new ProcedureInParameter("p_srec", vSrec, OracleTypes.STRUCT,
								"RECRV3_WRAPPER.PL_STD_CONTRL_AIRSPACE_COMM_TYPE"),
						new ProcedureOutParameter("p_err", OracleTypes.VARCHAR),
						new ProcedureOutParameter("p_valind", OracleTypes.VARCHAR),
						new ProcedureInParameter("p_view_only", pIgnoreRef, OracleTypes.VARCHAR));
				vNerr = toString(dbCalls.get("p_err"));
				if (!Objects.equals(vNerr, "YYY")) {
					if (Objects.equals(substr(vNerr, 1, 1), N)) {
						vAllErr = getNullClean(vAllErr) + " * 461 - " + coreptLib.getErrText(461);

					}

					if (Objects.equals(substr(vNerr, 2, 1), N)) {
						vAllErr = getNullClean(vAllErr) + " * 462 - " + coreptLib.getErrText(462);

					}

					if (Objects.equals(substr(vNerr, 3, 1), N)) {
						vAllErr = getNullClean(vAllErr) + " * 463 - " + coreptLib.getErrText(463);

					}

				}

				coreptLib.setindandmessage(pBlock, vAllErr, parameter.getWorkType(), pIgnoreRef, vInd);
				if (Objects.equals(pIgnoreRef, N)) {

					setUpdateDcr();

				}

				coreptLib.setoverridebutton(pBlock);

			}

			else {
coreptLib.setinitialerrordisplay(pBlock);

			}

			log.info("doValidate Executed Successfully");
		} catch (Exception e) {
			log.error("Error while executing doValidate{}", e.getMessage());
			throw e;

		}

	@Override
	public Map<String, Object> populateRelRecord(String pBlock) throws Exception {
		log.info("populateRelRecord Executing");
		Map<String, Object> result = new HashMap<>();
		PlStdContrlAirspaceCommType pSrec = new PlStdContrlAirspaceCommType();
		try {

			if (Objects.equals(upper(pBlock), DbConstants.PL_STD_CONTRL_AIRSPACE_COMM_MR)) {
				pSrec.setCenterIcaoCode(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCenterIcaoCode());
				pSrec.setMultCode(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getMultCode());
				pSrec.setAirspaceClassification(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirspaceClassification());
				pSrec.setAirportIcao(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIcao());
				pSrec.setAirportIdent(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent());
				pSrec.setFirUirAddress(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirAddress());
				pSrec.setFirUirIdent(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirIdent());
				pSrec.setFirUirInd(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirInd());
				pSrec.setAirspaceCenter(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirspaceCenter());
				pSrec.setAirspaceType(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirspaceType());
				pSrec.setRemoteName(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getRemoteName());
				pSrec.setCommunicationType(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCommunicationType());
				pSrec.setFreqUnits(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFreqUnits());

				pSrec.setCommunicationFreq(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
						.getCommunicationFreq()!=null ?  plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
						.getCommunicationFreq().getDoubleValue():null);
				pSrec.setValidateInd(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getValidateInd());
				if (toInteger(global.getRecentCycle()) >= toInteger(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getProcessingCycle())) {
					pSrec.setProcessingCycle(toInteger(
							plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getProcessingCycle()));

				}

				else {
					pSrec.setProcessingCycle(toInteger(global.getRecentCycle()));

				}
				pSrec.setDataSupplier(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getDataSupplier());
				pSrec.setCreateDcrNumber(toInteger(
						plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCreateDcrNumber()));

			}
			result.put("pSrec", pSrec);

			log.info("populateRelRecord Executed Successfully");
			return result;
		} catch (Exception e) {
			log.error("Error while executing populateRelRecord{}", e.getMessage());
			throw e;

		}
	}

	@Override
	public void setUpdateDcr() throws Exception {
		log.info("setUpdateDcr Executing");
		String query = "";
		try {
			PlStdContrlAirspaceCommType rstdContrlAirspaceComm;
			Integer vDcr = toInteger(
					plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCreateDcrNumber());
			Integer vCount = 0;

			query = "select * from pl_std_heli_path_point where create_dcr_number = ?";
			List<DuRecord> rstdApproachs = app.executeQuery(query, vDcr);

			for (DuRecord rstdHeliPathPoints : rstdApproachs) {
				rstdContrlAirspaceComm = app.mapResultSetToClass(rstdHeliPathPoints, PlStdContrlAirspaceCommType.class);

				vCount = vCount + 1;
				if (!Objects.equals(
						nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getCommunicationType(),
								" "),
						nvl(rstdContrlAirspaceComm.getCommunicationType(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
										.getCommunicationClass(), " "),
								nvl(rstdContrlAirspaceComm.getCommunicationClass(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
										.getCommunicationFreq(), 0),
								nvl(rstdContrlAirspaceComm.getCommunicationFreq(), 0))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFreqUnits(),
										" "),
								nvl(rstdContrlAirspaceComm.getFreqUnits(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getRemoteName(),
										" "),
								nvl(rstdContrlAirspaceComm.getRemoteName(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirIdent(),
										" "),
								nvl(rstdContrlAirspaceComm.getFirUirIdent(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirAddress(),
										" "),
								nvl(rstdContrlAirspaceComm.getFirUirAddress(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getFirUirInd(),
										" "),
								nvl(rstdContrlAirspaceComm.getFirUirInd(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIdent(),
										" "),
								nvl(rstdContrlAirspaceComm.getAirportIdent(), " "))
						|| !Objects.equals(
								nvl(plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex()).getAirportIcao(),
										" "),
								nvl(rstdContrlAirspaceComm.getAirportIcao(), " "))) {
					plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
							.setUpdateDcrNumber(new CustomInteger(global.getDcrNumber()));

				}

			}
			if (Objects.equals(vCount, 0)) {
				plStdContrlAirspaceCommMr.getRow(system.getCursorRecordIndex())
						.setUpdateDcrNumber(new CustomInteger(global.getDcrNumber()));

			}

			log.info("setUpdateDcr Executed Successfully");
		} catch (Exception e) {
			log.error("Error while executing setUpdateDcr{}", e.getMessage());
			throw e;

		}
	}

	@Override
	public ResponseEntity<ResponseDto<ContrlAirspaceCommTriggerResponseDto>> onError(
			ContrlAirspaceCommTriggerRequestDto reqDto) throws Exception {
		log.info(" onError Executing");
		BaseResponse<ContrlAirspaceCommTriggerResponseDto> responseObj = new BaseResponse<>();
		ContrlAirspaceCommTriggerResponseDto resDto = new ContrlAirspaceCommTriggerResponseDto();
		try {
			OracleHelpers.bulkClassMapper(reqDto, this);
			Integer msgnum = global.getErrorCode();
			String msgtxt = global.getErrorText();
			String msgtyp = global.getErrorType();
			String vBlockName = system.getCursorBlock();

			if ((Objects.equals(msgnum, 40400) || Objects.equals(msgnum, 40406) || Objects.equals(msgnum, 40407))) {
				message("Changes saved successfully");

			}

			else if (Arrays.asList(41051, 40350, 47316, 40353).contains(msgnum)) {
				log.debug("msgnum value is {}", msgnum);
			}

			String pBlock = toSnakeCase(system.getCursorBlock());
			Integer vAllowUpdate = 0;

			if (like(PL_UNDERSCORE_PATTERN, pBlock)) {

						&& !Objects.equals(nameIn(this, pBlock + ".processing_cycle"), null)) {

					doValidate(pBlock, "");

				}

				else {

					coreptLib.setinitialerrordisplay(pBlock);

				}
				if (!Objects.equals(parameter.getWorkType(), "VIEW")
						&& Objects.equals(parameter.getRecordType(), T)) {
					if (Objects.equals(nameIn(this, pBlock + ".generated_in_house_flag"), N)) {
						setBlockItemProperty(pBlock + ".processing_cycle", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_FALSE);
						setBlockItemProperty(pBlock + ".generated_in_house_flag", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_FALSE);
						setBlockItemProperty(pBlock + ".validate_ind", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_TRUE);
						if (!like(PATTERN_MR, pBlock)) {
							setItemProperty(pBlock + ".processing_cycle", "current_record_attribute", "Not_updatable");
							setItemProperty(pBlock + ".generated_in_house_flag", "current_record_attribute",
									"Not_updatable");
							setItemProperty(pBlock + ".validate_ind", "current_record_attribute", "default");

						}

					}

					else {
						setBlockItemProperty(pBlock + ".generated_in_house_flag", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_TRUE);
						setBlockItemProperty(pBlock + ".validate_ind", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_FALSE);
						if (!like(PATTERN_MR, pBlock)) {
							setItemProperty(pBlock + ".generated_in_house_flag", "current_record_attribute", "default");
							setItemProperty(pBlock + ".validate_ind", "current_record_attribute", "Not_updatable");

						}

								&& Objects.equals(nameIn(this, pBlock + ".generated_in_house_flag"), Y)) {
							setBlockItemProperty(pBlock + ".processing_cycle", FormConstant.UPDATE_ALLOWED,

							}

						}

						else {
							setBlockItemProperty(pBlock + ".processing_cycle", FormConstant.UPDATE_ALLOWED,

		for (Integer index : currentRecordIndexes) {

	public void validateBeforeCommit(ContrlAirspaceCommTriggerRequestDto reqDto) throws Exception {
		int oldIndex = system.getCursorRecordIndex();
		DuBlock<?> recs = ((DuBlock<?>) nameIn(this, system.getCursorBlock()));
		system.setCursorRecordIndex(0);
		for (Object rec : recs.getData()) {
			if (Objects.equals(nameIn(rec, "recordStatus"), FormConstant. INSERT)) {
				plStdContrlAirspaceCommMrPreInsert(reqDto);
				whenValidateRecord(reqDto);
			}
			system.setCursorRecordIndex(system.getCursorRecordIndex() + 1);
		}
		system.setCursorRecordIndex(0);
		for (Object rec : recs.getData()) {
			if (Objects.equals(nameIn(rec, "recordStatus"), FormConstant. CHANGED)) {
				plStdContrlAirspaceCommMrPreUpdate(reqDto);
				whenValidateRecord(reqDto);
			}
			system.setCursorRecordIndex(system.getCursorRecordIndex() + 1);
		}
		system.setCursorRecordIndex(oldIndex);
	}

		BaseResponse<ContrlAirspaceCommTriggerResponseDto> responseObj = new BaseResponse<>();
		ContrlAirspaceCommTriggerResponseDto resDto = new ContrlAirspaceCommTriggerResponseDto();
		OracleHelpers.bulkClassMapper(reqDto, this);

		try {

			String pBlock = toSnakeCase(system.getCursorBlock());

			if (like(PL_UNDERSCORE_PATTERN, pBlock) && Arrays.asList(FormConstant. CHANGED, FormConstant. INSERT).contains(system.getRecordStatus())) {

						setBlockItemProperty(pBlock + ".validate_ind", FormConstant.UPDATE_ALLOWED,

								FormConstant.PROPERTY_TRUE);
						setBlockItemProperty(pBlock + ".generated_in_house_flag", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_TRUE);
						setBlockItemProperty(pBlock + ".validate_ind", FormConstant.UPDATE_ALLOWED,
								FormConstant.PROPERTY_FALSE);
						if (like(PATTERN_MR, pBlock)) {
							setItemProperty(pBlock + ".processing_cycle", "current_record_attribute", "default");
							setItemProperty(pBlock + ".generated_in_house_flag", "current_record_attribute", "default");
							setItemProperty(pBlock + ".validate_ind", "current_record_attribute", "notUpdatable");

						}

				}

	@Override
	public ResponseEntity<ResponseDto<ExportDestinationTriggerResponseDto>> exportDestination(
			ExportDestinationTriggerRequestDto reqDto) throws Exception {
		log.info(" exportDestination Executing");
		BaseResponse<ExportDestinationTriggerResponseDto> responseObj = new BaseResponse<>();
		ExportDestinationTriggerResponseDto resDto = new ExportDestinationTriggerResponseDto();
		try {
			OracleHelpers.bulkClassMapper(reqDto, this);
			BlockDetail mstBlockData = null;
			StringBuilder reportfile = new StringBuilder();
			List<DuRecord> recs = null;
			if (OracleHelpers.isNullorEmpty(selectOptions.getFileName())) {
				PropertyHelpers.setAlertProperty(event, "dsp_msg", "stop", "Forms",
						" Client file name cannot be null", "ALERT_MESSAGE_TEXT", "OK", null, null);
				PropertyHelpers.setShowAlert(event, "dsp_msg", false);
				throw new FormTriggerFailureException();
			}

			if (HoneyWellUtils.toCamelCase(system.getCursorBlock()).equals("plStdContrlAirspaceCommMr")) {
				mstBlockData = reqDto.getExportDataBlocks().get("plStdContrlAirspaceCommMr");
				List<String> mstPromptNames = getBlockMetaData(mstBlockData, "PROMPT_NAME");
				List<String> mstDatabseColumns = getBlockMetaData(mstBlockData, "DATABASE_COLUMN");
				String query = hashUtils.decrypt(mstBlockData.getLastQuery());
				if (query.contains(" where rno <=")) {
					query = substrWithLen(query, 0, OracleHelpers.instr(query, " where rno <="));
				} else if (query.contains(" OFFSET ")) {
					query = substrWithLen(query, 0, OracleHelpers.instr(query, " OFFSET "));
				}
				reportfile.append(getExportHeader(mstPromptNames, 0, selectOptions.getDelimiter()));
				// Master Fetching..
				recs = app.executeQuery(query);
				for (DuRecord mstRec : recs) {
					PlStdContrlAirspaceCommMr plStdContrlAirspaceCommMr = app.mapResultSetToClass(mstRec,
							PlStdContrlAirspaceCommMr.class);
					reportfile.append(getExportData(plStdContrlAirspaceCommMr, mstDatabseColumns, 0,
							selectOptions.getDelimiter(), selectOptions.getGetTextFile()));
				}

			} else if (HoneyWellUtils.toCamelCase(system.getCursorBlock()).equals("stdContrlAirspaceComm")) {
				mstBlockData = reqDto.getExportDataBlocks().get("stdContrlAirspaceComm");
				List<String> mstPromptNames = getBlockMetaData(mstBlockData, "PROMPT_NAME");
				List<String> mstDatabseColumns = getBlockMetaData(mstBlockData, "DATABASE_COLUMN");
				String query = hashUtils.decrypt(mstBlockData.getLastQuery());
				if (query.contains(" where rno <=")) {
					query = substrWithLen(query, 0, OracleHelpers.instr(query, " where rno <="));
				} else if (query.contains(" OFFSET ")) {
					query = substrWithLen(query, 0, OracleHelpers.instr(query, " OFFSET "));
				}
				reportfile.append(getExportHeader(mstPromptNames, 0, selectOptions.getDelimiter()));
				// Master Fetching..
				recs = app.executeQuery(query);
				for (DuRecord mstRec : recs) {
					StdContrlAirspaceComm stdContrlAirspaceComm = app.mapResultSetToClass(mstRec,
							StdContrlAirspaceComm.class);
					reportfile.append(getExportData(stdContrlAirspaceComm, mstDatabseColumns, 0,
							selectOptions.getDelimiter(), selectOptions.getGetTextFile()));
				}
			}

			else if (HoneyWellUtils.toCamelCase(system.getCursorBlock()).equals("message")) {
				List<String> messageLogs = new ArrayList<>();
				String dateQuery = """
						     SELECT TO_CHAR(SYSDATE , 'Month DD,YYYY') as formatted_date  FROM DUAL
						""";

				String timeQuery = """
						SELECT  to_char(sysdate,'HH24:MI') FROM DUAL
						""";
				DuRecord dateRec = app.selectInto(dateQuery);
				DuRecord timeRec = app.selectInto(timeQuery);

				reportfile.append("Generated on ").append(dateRec.getObject()).append(" at ")
						.append(timeRec.getObject()).append("\n").append("\n");
				mstBlockData = reqDto.getExportDataBlocks().get("message");
				messageLogs = mstBlockData.getMessageLogs();
				for (int i = 0; i <= messageLogs.size() - 1; i++) {
					reportfile.append(messageLogs.get(i)).append("\n");
				}
			} else if (HoneyWellUtils.toCamelCase(system.getCursorBlock()).equals("message")) {
				List<String> messageLogs = new ArrayList<>();
				String dateQuery = """
						     SELECT TO_CHAR(SYSDATE , 'Month DD,YYYY') as formatted_date  FROM DUAL
						""";

				String timeQuery = """
						SELECT  to_char(sysdate,'HH24:MI') FROM DUAL
						""";
				DuRecord dateRec = app.selectInto(dateQuery);
				DuRecord timeRec = app.selectInto(timeQuery);

				reportfile.append("Generated on ").append(dateRec.getObject()).append(" at ")
						.append(timeRec.getObject()).append("\n").append("\n");
				mstBlockData = reqDto.getExportDataBlocks().get("message");
				messageLogs = mstBlockData.getMessageLogs();
				for (int i = 0; i <= messageLogs.size() - 1; i++) {
					reportfile.append(messageLogs.get(i)).append("\n");
				}
			}
			OracleHelpers.ResponseMapper(this, resDto);
			String base64 = Base64.getEncoder().encodeToString(reportfile.toString().getBytes(StandardCharsets.UTF_8));
			ReportDetail reportDetail = new ReportDetail();
			reportDetail.setData(base64);
			resDto.setReport(reportDetail);
			goBlock(system.getCursorBlock(), "firstItem");
			log.info(" exportDestination executed successfully");
			return responseObj.render(responseObj.formSuccessResponse(DuConstants.RECORD_FETCH, resDto));
		} catch (RuntimeException e) {
			log.info("RuntimeException captured");
			OracleHelpers.ResponseMapper(this, resDto);
			return ExceptionUtils.handleException(e, resDto);
		} catch (Exception e) {
			log.error("Error while Executing the exportDestination Service");
			OracleHelpers.ResponseMapper(this, resDto);
			return ExceptionUtils.handleException(e, resDto);
		}

	}
