PROCEDURE check_to_commit (p_action_type VARCHAR2)
IS
   v_button        NUMBER (1);
   total_rows      NUMBER;
   v_temp          VARCHAR2 (4000);
   v_button_text   VARCHAR2 (100);
   msgnum          NUMBER          := MESSAGE_CODE; -- Fix - SCR 5180 - Eswar KC - 02-Sep-2011
BEGIN
   IF :SYSTEM.form_status = 'CHANGED'
   THEN
      IF p_action_type = 'COMMIT' THEN
         v_button_text := 'Cancel';
      ELSIF p_action_type = 'EXIT' THEN
         v_button_text := 'Exit Without Save';
      ELSE
         v_button_text := 'Cancel Modification';
      END IF;

      IF :GLOBAL.lib_refreshed = 'Y' THEN
         v_button :=
            display_alert.more_buttons
                       ('S','Refresh Record','You have modified record(s). Select an option: '|| CHR (10)|| ' '|| CHR (10)|| 
                         --'1. Save without refreshing the Master Library'||chr(10)|| -- Fix - SCR 5180 - Eswar KC - 02-Sep-2011
                           '1. Save and refresh Master Library'|| CHR (10)|| 
						   						 '2. Cancel modification, NO Save, NO Refresh',
                           'Save&Refresh',v_button_text
                        );

         IF v_button IN (1) THEN -- changed from in (1,2) to in (1) -- Fix - SCR 5180 - Eswar KC - 02-Sep-2011
            COMMIT_FORM;
            total_rows :=
                   GET_GROUP_ROW_COUNT (FIND_GROUP ('Refresh_Records_Group'));

            IF total_rows > 0 THEN
               refresh_master_library.refresh_records (total_rows);
            END IF;

            IF (msgnum = 40400) THEN   -- Fix - SCR 5180 - Eswar KC - 02-Sep-2011
               CLEAR_MESSAGE;
            END IF;
         END IF;
      ELSE
         IF p_action_type = 'COMMIT' THEN
            v_button := 1;
         ELSE
            v_button :=
               display_alert.more_buttons
                  ('S','Refresh Record','Record is modified or inserted or deleted. Select an option: '|| CHR (10)|| ' ',
                   'Save',v_button_text
                  );
         END IF;

-- Begin Fix - SCR 5180 - Eswar KC - 02-Sep-2011
         IF v_button = 1 THEN
            COMMIT_FORM;

            IF (msgnum = 40400) THEN
               CLEAR_MESSAGE;
            END IF;
         ELSE
            EXIT_FORM (no_commit);
         END IF;
-- End Fix - SCR 5180 - Eswar KC - 02-Sep-2011
      END IF;

      -- Here the v_button values are chnaged 2 in place of 3, 1 in place of 2 for SCR 5180 - Eswar KC
      IF    (:GLOBAL.lib_refreshed = 'Y' AND v_button = 2)
         OR (:GLOBAL.lib_refreshed != 'Y' AND v_button = 1) THEN
         IF p_action_type = 'COMMIT' THEN
            NULL;
         ELSIF p_action_type = 'EXIT' THEN
            SET_APPLICATION_PROPERTY (cursor_style, 'DEFAULT');
            EXIT_FORM (no_commit);
         ELSE
            CLEAR_BLOCK (no_commit);
         END IF;
      ELSE
         IF :SYSTEM.form_status = 'CHANGED' THEN
            RAISE form_trigger_failure;
         END IF;
      END IF;
   --if :system.form_status != 'CHANGED' then
   -- invalid_record_count(:system.cursor_block,:system.cursor_record,:system.cursor_item);
   --end if;
   ELSE
      refresh_master_library.if_refresh;
   END IF;
END;

PROCEDURE do_validate(p_block varchar2,p_ignore_ref varchar2 := 'Y') IS
  v_rrecord  sdv_records.cr_restr_airspace_comm;       
  v_crecord  sdv_records.cr_contrl_airspace_comm;
  v_srec    PL_STD_CONTRL_AIRSPACE_COMM%rowtype;       
	v_err_ind	number := 0;
	v_err_list cpt_types.errlist_type;
	v_all_err varchar2(4000) := null;
	v_nerr    varchar2(3);
	v_valid   varchar2(1) := null;
	v_class   varchar2(1);
	v_ind     varchar2(1) := 'Y';
BEGIN
	if :PL_STD_CONTRL_AIRSPACE_COMM_MR.AIRSPACE_TYPE is not null then 
		:CONTROL_BLOCK.STD_OVERRIDE_ERRORS := null;
		v_class := :pl_std_contrl_airspace_comm_mr.communication_class;
		if v_class not in ('E','A') then
			if not is_override(:global.data_supplier, :PL_STD_CONTRL_AIRSPACE_COMM_MR.processing_cycle, 'CONTRL_AIRSPACE_COMM', 979) then
				v_all_err := '* 979 - '||get_err_text(979);		-- '* Communication Class must be ''A'' or ''E''.';
				v_ind := 'I';
			else
				if v_ind != 'I' then
					v_ind := 'O';
				end if;
			end if;	
			if (rtrim(:pl_std_contrl_airspace_comm_mr.airport_ident) is null and
				rtrim(:pl_std_contrl_airspace_comm_mr.fir_uir_ident) is null) or
				(rtrim(:pl_std_contrl_airspace_comm_mr.airport_ident) is not null and
				rtrim(:pl_std_contrl_airspace_comm_mr.fir_uir_ident) is not null)then
				if not is_override(:global.data_supplier, :PL_STD_CONTRL_AIRSPACE_COMM_MR.processing_cycle, 'CONTRL_AIRSPACE_COMM', 980) then
					v_all_err := v_all_err||' * 980 - '||get_err_text(980);
					--' * The recorcd shall have either Airport Communication data only or Enroute Communication data only.';
					v_ind := 'I';
				else
					if v_ind != 'I' then
						v_ind := 'O';
					end if;
				end if;	
			end if;
		elsif v_class = 'A' and 
  		(rtrim(:pl_std_contrl_airspace_comm_mr.airport_ident) is null or
  	 	rtrim(:pl_std_contrl_airspace_comm_mr.fir_uir_ident) is not null) then
  	  if not is_override(:global.data_supplier, :PL_STD_CONTRL_AIRSPACE_COMM_MR.processing_cycle, 'CONTRL_AIRSPACE_COMM', 981) then
				v_all_err := '* 981 - '||get_err_text(981);
  	 		-- '* Communiation Class is ''A'', the record shall only have Airport Communication data. ';
				v_ind := 'I';
		  else
				if v_ind != 'I' then
					v_ind := 'O';
				end if;
			end if;	  	 
  	elsif v_class = 'E' and
  		(rtrim(:pl_std_contrl_airspace_comm_mr.airport_ident) is not null or
  	 	rtrim(:pl_std_contrl_airspace_comm_mr.fir_uir_ident) is null) then
  	  if not is_override(:global.data_supplier, :PL_STD_CONTRL_AIRSPACE_COMM_MR.processing_cycle, 'CONTRL_AIRSPACE_COMM', 982) then
				v_all_err := '* 982 - '||get_err_text(982);
  	 		--'* Communiation Class is ''E'', the record shall only have Enroute Communication data. ';
				v_ind := 'I';
		  else
				if v_ind != 'I' then
					v_ind := 'O';
				end if;
		  end if;	  	   	 
  	end if;
		populate_rel_record(p_block, v_srec);
		RECRV3.vcontrl_airspace_comm('S',v_srec,v_nerr,v_valid, p_ignore_ref);
		if v_nerr != 'YYY' then
  		if substr(v_nerr,1,1) = 'N' then
  			v_all_err := v_all_err||' * 461 - '||get_err_text(461);
  		end if;
  		if substr(v_nerr,2,1) = 'N' then
  			v_all_err := v_all_err||' * 462 - '||get_err_text(462);
  		end if;
  		if substr(v_nerr,3,1) = 'N' then
  			v_all_err := v_all_err||' * 463 - '||get_err_text(463);
  		end if;
  	end if;
  	-- Set validate_ind and display message
  	Set_Ind_and_Message(p_block,v_all_err, :parameter.work_type, p_ignore_ref, v_ind);
  	if p_ignore_ref = 'N' then
  		set_update_dcr;
  	end if;	
  	Set_Override_Button(p_block);
  ELSE
  	set_initial_error_display(P_BLOCK);
  end if;
END;

 setUpdateDcr:
PROCEDURE SET_UPDATE_DCR IS
	get_rec_cur  cpt_types.refcur;
	rstd_CONTRL_airspace_comm PL_STD_CONTRL_AIRSPACE_COMM%rowtype;
	v_dcr        number      := :PL_STD_CONTRL_AIRSPACE_COMM_MR.create_dcr_number;
	v_count      number      := 0;
BEGIN
		open get_rec_cur for select * from PL_STD_CONTRL_AIRSPACE_COMM
							           where create_dcr_number = v_dcr;	
		loop
		fetch get_rec_cur into rstd_CONTRL_airspace_comm;
		exit when get_rec_cur%notfound;
		  v_count := v_count + 1;
		  if nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.COMMUNICATION_TYPE,' ')  != nvl(rstd_CONTRL_airspace_comm.COMMUNICATION_TYPE,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.COMMUNICATION_CLASS,' ') != nvl(rstd_CONTRL_airspace_comm.COMMUNICATION_CLASS,' ')   or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.COMMUNICATION_FREQ,0)    != nvl(rstd_CONTRL_airspace_comm.COMMUNICATION_FREQ,0) or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.FREQ_UNITS,' ')          != nvl(rstd_CONTRL_airspace_comm.FREQ_UNITS,' ') or 
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.REMOTE_NAME,' ')         != nvl(rstd_CONTRL_airspace_comm.REMOTE_NAME,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.FIR_UIR_IDENT,' ')       != nvl(rstd_CONTRL_airspace_comm.FIR_UIR_IDENT,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.FIR_UIR_ADDRESS,' ')     != nvl(rstd_CONTRL_airspace_comm.FIR_UIR_ADDRESS,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.FIR_UIR_IND,' ')         != nvl(rstd_CONTRL_airspace_comm.FIR_UIR_IND,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.AIRPORT_IDENT,' ')       != nvl(rstd_CONTRL_airspace_comm.AIRPORT_IDENT,' ') or
				 nvl(:PL_STD_CONTRL_AIRSPACE_COMM_MR.AIRPORT_ICAO,' ')        != nvl(rstd_CONTRL_airspace_comm.AIRPORT_ICAO,' ') then
         :PL_STD_CONTRL_AIRSPACE_COMM_MR.update_dcr_number := :global.dcr_number;  -- Fix - SCR 5175 - Eswar KC - o6 July 2011
		  end if;  
		end loop;
		close get_rec_cur;
		if v_count = 0 then   -- it is a new record
			:PL_STD_CONTRL_AIRSPACE_COMM_MR.update_dcr_number := :global.dcr_number;  -- Fix - SCR 5175 - Eswar KC - o6 July 2011
		end if;
END;

					v_validate_ind := refresh_ml_utilities.get_validate_ind(get_block_property(v_block_name,QUERY_DATA_SOURCE_NAME),
																							:PL_STD_CONTRL_AIRSPACE_COMM_MR.create_dcr_number);
				end if;
				--
    		if v_status != 'NEW' then
    			if v_validate_ind in ('Y','S','H','O') then
    				IF refresh_master_library.check_reference_info(v_block_name,'D') = 'N' THEN
    					raise form_trigger_failure;
						END IF;																						
					end if;
					--
					IF :global.lib_refreshed = 'Y' and
						v_processing_cycle in (:global.new_processing_cycle,:global.old_processing_cycle) and -- Fix - SCR 0024 - Dass - 03-jan-2012
						v_validate_ind IN ('Y','S','H','O') THEN
						refresh_master_library.refresh_a_record(v_table_type,v_dcr_number,v_processing_cycle,ltrim(upper(get_block_property(v_block_name,QUERY_DATA_SOURCE_NAME)),'PL_'),'I');-- Fix - SCR 0024 - Dass - 03-jan-2012
						delete_record;-- Fix - SCR 0024 - Dass - 03-jan-2012
						commit_form;-- Fix - SCR 0024 - Dass - 03-jan-2012
						refresh_master_library.set_record_group(v_dcr_number,'I',v_block_name,v_processing_cycle,'U');
						dsp_msg('The refresh master library table for this deletion is done and all changes are commited.');-- Fix - SCR 0024 - Dass - 03-jan-2012
					else -- Fix - SCR 0024 - Dass - 03-jan-2012
						delete_record;-- Fix - SCR 0024 - Dass - 03-jan-2012
					END IF;
					--
					if v_validate_ind in ('Y','S','H','O') then
						refresh_master_library.delete_from_ref_table(v_dcr_number);
					end if;
				end if;									
			END;
		end if;
		else
		if :PL_STD_CONTRL_AIRSPACE_COMM_MR.validate_ind in ('Y','S','H','O') then
			refresh_master_library.delete_from_ref_table(:PL_STD_CONTRL_AIRSPACE_COMM_MR.create_dcr_number);
		end if;
		delete_record;
	end if;	
END IF;
